<h2>Schema of table </h2>
<hr>
CREATE TABLE Movie(
MOV_ID INT NOT NULL PRIMARY KEY,
TITLE VARCHAR(100),
Genere VARCHAR(50),
Director VARCHAR(50),
ACTOR VARCHAR(50),
ACTRESS VARCHAR(50),
RELEASE_YEAR INT,
LANGUAGE VARCHAR(30),
DURATION INT,
RATING FLOAT,
BUDGET BIGINT,
COLLECTION BIGINT
);
